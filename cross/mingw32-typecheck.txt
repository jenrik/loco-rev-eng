# Cross file for the MinGW typecheck build — NOT a shippable target.
#
# This compiles the same unified source tree used by the native Linux build
# with the i686-w64-mingw32 cross-compiler, which provides real MSVC-era
# Windows headers (windows.h, ddraw.h, dsound.h, dplay.h). Since _WIN32 is
# defined here, every #ifndef _WIN32-guarded host-only file compiles down to
# an empty translation unit for free — only the real decompiled classes get
# typechecked against real Windows headers. Equivalent to the old
# `make -C src/decompiled_cpp MINGW=1`.
#
# Usage:
#   meson setup build-mingw --cross-file cross/mingw32-typecheck.txt
#   meson compile -C build-mingw
#
# Available via: nix-shell -p pkgsCross.mingw32.stdenv.cc (or the project's
# default devShell, which already provides it as `mingwToolchain`).

[binaries]
c = 'i686-w64-mingw32-gcc'
cpp = 'i686-w64-mingw32-g++'
ar = 'i686-w64-mingw32-ar'
strip = 'i686-w64-mingw32-strip'
pkg-config = 'i686-w64-mingw32-pkg-config'

[host_machine]
system = 'windows'
cpu_family = 'x86'
cpu = 'i686'
endian = 'little'

[properties]
needs_exe_wrapper = true
